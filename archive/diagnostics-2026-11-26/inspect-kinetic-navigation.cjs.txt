const fs=require('fs');const {chromium}=require('@playwright/test');
(async()=>{const c=JSON.parse(fs.readFileSync('kinetic-browser.local.json'));const b=await chromium.launch({headless:false});const ctx=await b.newContext({storageState:c.storageState});await ctx.route('**/*',r=>c.origins.includes(new URL(r.request().url()).origin)?r.continue():r.abort());const pending=[];const out='captures/kinetic-navigation';fs.mkdirSync(out,{recursive:true});ctx.on('response',r=>{if(r.url().includes('/api/') || r.url().includes('CustShip'))pending.push((async()=>{let data;try{data=require('./discover-shipment.cjs').body(await r.text())}catch{}fs.appendFileSync(out+'/traffic.jsonl',JSON.stringify({method:r.request().method(),url:require('./discover-shipment.cjs').urlSafe(r.url()),status:r.status(),data})+'\n')})());});const p=await ctx.newPage();await p.goto(c.url);await p.waitForResponse(r=>r.url().includes('GetMenuForLicenseType'));await p.goto(c.inspectionUrl || c.url);await p.waitForFunction(()=>document.body.innerText.includes('Customer Shipment'),null,{timeout:15000}).catch(()=>{});for(const [i,f] of p.frames().entries()){fs.writeFileSync(out+'/frame-'+i+'.txt',await f.locator('body').innerText());fs.writeFileSync(out+'/frame-'+i+'.html',await f.locator('body').innerHTML());}console.log('Pages',ctx.pages().length,'Frames',p.frames().length);await Promise.allSettled(pending);await b.close();})().catch(e=>{console.log(e.message.split('\n')[0]);process.exitCode=1});
