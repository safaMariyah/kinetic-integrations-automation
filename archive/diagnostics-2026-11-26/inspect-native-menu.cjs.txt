}finally{for(const[j,page]of ctx.pages().entries())for(const[i,f]of page.frames().entries()){try{fs.writeFileSync(`${out}/page-${j}-frame-${i}.txt`,await f.locator('body').innerText());fs.writeFileSync(`${out}/page-${j}-frame-${i}.html`,await f.locator('body').innerHTML());}catch{}}await Promise.allSettled([...pending]);await ctx.close();await Promise.allSettled([...pending]);fs.writeFileSync(out+'/events.json',JSON.stringify(events,null,2));await b.close();}})().catch(e=>{console.error(e.message.split('\n')[0]);process.exitCode=1});
